الرحيق المختوم — باب ۲
عَرَبَن منٛز حُکمرٲنی تہٕ شہزادگی
صَفہٕ ۱۴–۲۱ · نظرِ ثٲنی خٲطرٕ

عَرَبَن ہٕنٛز حُکمرٲنی

اسلامَس برونٛہہ عَرَبَن ہٕنٛز کتھ کَرنہٕ وِزِ چھُ تِمَن ہٕندٮ۪ن حُکمرانَن، شہزادَن، قبیلَن ہٕنٛزَن تَقسیمَن تہٕ مذہبی رَہنماہن ہُنٛد مُختَصر جائزٕ پَیش کَرُن ضروٗری۔ اَمہِ سۭتۍ چھُ اَسہِ تِم حالات سَمَجھنَس منٛز مَدَتھ سَپدان، یِمن منٛز اسلام آو سوٗزنہٕ۔
[صَفہٕ 14]

ییلہِ اسلامُک سورُے وۄتھ، عَرَب کۍ حُکمران ٲسۍ دۄن قِسمَن ہٕندۍ۔ اَکھ ٲسۍ تاج پوش بادشاہ، یِم حَقیقَتَس منٛز آزاد ٲسۍ نہٕ۔ بیٚیہِ ٲسۍ قبیلَن تہٕ خاندٲنی ٹولن ہٕندۍ رَہنما، یِمَن تاج پوش بادشاہن ہِشے اِختیار تہٕ خٲص حَق حٲصِل ٲسۍ۔ یِم ٲسۍ زیادہٕ تَر آزاد، حالانٛکہِ یِمو منٛز کینہہ ٲسۍ کٲنسہِ تاج پوش بادشاہ سٕنٛز فرمانبردٲری مانان۔ تاج پوش بادشاہ ٲسۍ صِرف یمن، حیرہ تہٕ غَسّان کۍ۔ باقی سٲری عَرَب حُکمران ٲسۍ تاجٕ وَرٲے۔
[صَفہٕ 14]

یمنٕچ حُکمرٲنی

سَبا (Sheba) کۍ لوٗکھ ٲسۍ یمنَس منٛز روزن والٮ۪ن خالِص عَرَبَن ہٕنٛزَن ساروٕے کھۄتہٕ پرانۍ قَومَن منٛز اَکھ۔ «اور» (Or) نٲمِس جایہِ کھدٲیی سۭتۍ میول تہنٛدِ تقریباً ۲۵ صَدی قبل مسیح موٗجوٗد آسنُک پتاہ۔ تقریباً ۱۱مہِ صَدی قبل مسیحَس منٛز کٔر تِمَن ہٕنٛزِ تہذیبن ترقّی تہٕ پھیٚلیو تِمَن ہُنٛد مُلک۔
[صَفہٕ 14]

تِمَن ہُنٛد دَور ہٮ۪کَو اَمہِ اَندازٕ مُطٲبِق بانٹِتھ:
[صَفہٕ 14]

۱۔ ۶۵۰ قبل مسیح برونٛہہ کٮ۪ن صَدیَن منٛز اوس تِمَن ہٕندٮ۪ن بادشاہن «مُکَرِّب سَبا» وَننہٕ یِوان۔ تِمَن ہُنٛد دارالحکومت اوس صِرواح (Sarwah)، یَتھ خَریبہ تہِ وَنان ٲسۍ۔ اَمِکۍ کھنڈرات چھِ مارِب (Ma’rib) ہٕنٛدِ مَغرِبَس کُن اَکہِ دۄہَکِس پَیدَل سفرَس پؠٹھ۔ اَتھ دَورَس منٛز کٔر تِمو مارِب باندھ بَناوُن شروٗع، یَتھ یمنٕچ تٲریخَس منٛز بٔڈ اہمیت ٲس۔ سَبا سٕنٛز سلطنت چھےٚ تِتھ بٔڈ آسمٕژ وَننہٕ یِوان زِ تِمَن ہٕندۍ آباد علاقہٕ ٲسۍ عَرَبَس اَندر تہٕ نٮ۪بر تہِ۔

۲۔ ۶۵۰ قبل مسیح پیٹھٕ ۱۱۵ قبل مسیح تام تِمو تروو «مُکَرِّب» ناو، تہٕ لوگۍ پانَس «سَبا کۍ بادشاہ» وَنان۔ تِمو بَنٲو صِرواح بدلہٕ مارِب دارالحکومت۔ مارِب کۍ کھنڈرات چھِ صَنعاء (San‘a) ہٕنٛدِ مَشرِقَس کُن ۶۰ میٖل دُور۔

۳۔ ۱۱۵ قبل مسیح پیٹھٕ ۳۰۰ عیسوی تام کٔر حِمیَر قبیلَن سَبا ہٕنٛزِ بادشٲہی پؠٹھ قبضہٕ، تہٕ مارِب بدلہٕ بَنٲوُن رَیدان (Redan) دارالحکومت۔ رَیدانَس آو پَتہٕ ظَفار (Zifar) ناو تھٲونہٕ۔ اَمِکۍ کھنڈرات چھِ وُنہِ تہِ یَریم (Yarim) شہرس نِش مُدَوّر پہاڑَس پؠٹھ۔ اَتھ دَورَس منٛز ہٮ۪وت تِمَن ہُنٛد زوال شروٗع گژھُن۔ تِمَن ہٕنٛز تجارت گٔیہِ واریاہ کم؛ گوڈنیک سَبَب اوس شُمالی حجازَس پؠٹھ نَبطِیَن ہُنٛد قابوٗ، دوٚیم مصر، شام تہٕ شُمالی حجاز فتح کَرنہٕ پَتہٕ سمندری تجٲرتی وَتَن پؠٹھ رومیَن ہٕنٛز بَرتری، تہٕ ترٛٮ۪یم قبیلَن دَرمیان جَنٛگ۔ یِمن ترٛٮ۪ن سَبَبن کِنۍ گوٚو قحطان سٕندۍ خاندان الگ الگ تہٕ مُنتَشِر۔

۴۔ ۳۰۰ عیسوی پیٹھٕ یمنَس منٛز اسلامُک گاش وۄتھنَس تام اوس دَور واریاہ اَفراتفری تہٕ فِتنہٕ ہُنٛد۔ واریاہ جَنٛگَن تہٕ خانہٕ جَنٛگی کِنۍ آیہِ یمن کۍ لوٗکھ نیبرِمین ہٕندِس قابوٗس تَل تہٕ گٔیہِ تِمَن ہٕنٛز آزٲدی ختم۔ اَتھ دَورَس منٛز کٔر رومیَن عَدَن (Aden) فتح، تہٕ ہَمدان تہٕ حِمیَر دَرمیان لَگاتار جَنٛگُک فٲیدٕ تُلِتھ کٔرِکھ حَبَشہ کٮ۪ن لوٗکھن ۳۴۰ عیسوی منٛز گوڈنیک لَٹہِ یمنَس پؠٹھ قبضہٕ کَرنَس منٛز مَدَتھ۔ حَبَشہ ہُنٛد قبضہٕ روٗد ۳۷۸ عیسوی تام، تہٕ پَتہٕ گوٚو یمن دُبارٕ آزاد۔ پَتہٕ پَییہِ مارِب باندھَس منٛز دراڑٕ، یِمَن سۭتۍ آو ۴۵۰ یا ۴۵۱ عیسوی منٛز بٔڈ سیلاب، یَمہِ ہُنٛد ذِکر قرآنَس منٛز چھُ۔ یہِ اوس اَکھ بٔڈ واقعہٕ، یَمہِ کِنۍ یمنٕچ سورٕ تہذیب گٔیہِ زوالَس کُن تہٕ تَتِکۍ روزن والۍ لوٗکھ گوٚو مُنتَشِر۔
[صَفہٕ 14–15]

۵۲۳ عیسوی منٛز سوٗز یہوٗدی بادشاہ ذُو نُواسَن (Dhu Nawas) نَجران (Najran) کٮ۪ن عیسٲیَن خلاف اَکھ بٔڈ فوجی لشکر، یُتھ تِمَن زوٗرٕ سۭتۍ یہوٗدی مذہب قبول کَرناوِ۔ ییلہِ تِمو اَمہِ نِش اِنکار کوٚر، آیہِ تِم زِندٕ اَکِس بٔڑِس کھۄلَس منٛز ترٲونہٕ، یَتھ منٛز بٔڈ نار زٲلمُت اوس۔ قرآنَس منٛز چھُ اَمہِ واقعہٕ ہُنٛد اَمہِ پٲٹھۍ ذِکر: «لعنت چھےٚ کھۄلہِ والٮ۪ن پؠٹھ۔» [القرآن 85:4؛ کِتابہِ منٛز آمِتِس اَنٛگریزی مفہوٗمُک ترجمہٕ۔]
[صَفہٕ 15]

اَمہِ سۭتۍ آو عیسٲیَن، خاص پٲٹھۍ رومی شہنشاہَن، واریاہ غُصّہٕ۔ تِمو صِرف حَبَشہ کۍ لوٗکھ عَرَبَن خلاف اُکسٲوۍ نہٕ، بلکہ جمع کٔرِکھ بٔڈۍ سمندری جہازٕ تہِ۔ اَمہِ سۭتۍ میٚل حَبَشی فوجٕکٮ۪ن ۷۰٬۰۰۰ جَنٛگ کَرن والٮ۪ن مَدَتھ، تہٕ تِمو کٔر اَریاط (Eriat) سٕنٛزِ قیادتَس منٛز ۵۲۵ عیسوی منٛز یمن دوٚیمہِ لَٹہِ فتح۔ اَریاط آو یمنُک حُکمران بَناونہٕ۔ سُہ روٗد اَتھ عُہدَس پؠٹھ یوت تام پَننِس اَکِس فوجی رَہنماہَن، اَبرَہہَن (Abraha)، سُہ مور۔ اَبرَہہَن کٔر حَبَشہ کِس بادشاہَس سۭتۍ صُلح تہٕ نیو یمنُک اِقتِدار۔ پَتہٕ سوٗزُن کعبہٕ شَریف تَباہ کَرنہٕ خٲطرٕ پَنُن لشکر، تہٕ اَمہِ کِنۍ آیہِ سُہ تہٕ تَمہِ سٕندۍ سِپاہی «ہوٚس والۍ لوٗکھ»، یعنی اَصحابُ الفیٖل، ناوٕ زاننہٕ۔
[صَفہٕ 15]

ہوٚس کِس واقعہٕ پَتہٕ کٔر یمن کٮ۪ن لوٗکھن مَعدِی کَرِب بن سَیف ذی یَزَن الحِمیَری سٕنٛزِ قیادتَس منٛز، تہٕ فارسیَن ہٕنٛزِ مَدَتہٕ سۭتۍ حَبَشی حملہٕ آورَن خلاف بغاوت۔ تِمو حٲصِل کٔر دُبارٕ آزٲدی تہٕ بَنٲوُکھ مَعدِی کَرِب بادشاہ۔ مگر پَتہٕ مور سُہ اَکِس حَبَشی شَخصَن، یُس تٔمۍ پَننہِ خِدمت تہٕ حِفاظت خٲطرٕ پانَس نِش تھاومُت اوس۔ اَمہِ پٲٹھۍ گوٚو ذی یَزَن سُنٛد خاندان ہمیشہٕ خٲطرٕ بادشٲہی نِش مَحروٗم۔ فارسی بادشاہ کِسریَٰن تھاو صَنعاءَس پؠٹھ اَکھ فارسی حُکمران تہٕ اَمہِ پٲٹھۍ آو یمن فارسی قابوٗس تَل۔ فارسی حُکمران روٗدۍ یمنَس پؠٹھ راج کَران، یوت تام تِمَن ہٕندِ ٲخری حُکمران باذانَن (Badhan) اسلام قبول کوٚر۔ اَتھ اَنٛگریزی ایڈیشنَس منٛز چھُ اَمہِ خٲطرٕ ۶۳۸ عیسوی سَنہٕ دِنہٕ آمُت۔ اَمہِ سۭتۍ گٔیہِ یمنَس منٛز فارسی حکمرٲنی ختم۔
[صَفہٕ 15]

حیرہٕچ حُکمرٲنی

ییلہِ پیٹھٕ کوٗرُشِ اَعظمَن (Cyrus the Great، ۵۵۷–۵۲۹ قبل مسیح) فارسی اَکس کٔرۍ، تِم روٗدۍ عراق تہٕ اَمہِ زوٚپٲر کٮ۪ن علاقَن پؠٹھ حُکمرٲنی کَران۔ کانٛہہ ہیٚوک نہٕ تِمَن ہُنٛد قابوٗ ہَٹٲوِتھ، یوت تام اِسکَندرِ اَعظمَن فارسیَن ہٕندِس بادشاہَس شکست دِژ۔ اَتھ ایڈیشنَس منٛز چھُ بادشاہُک ناو «دارا اوّل» تہٕ سَنہٕ ۳۲۶ قبل مسیح دِنہٕ آمُت۔ اَمہِ پَتہٕ آیہِ فارسی زَمیٖنہٕ واریاہن حِصّن منٛز بانٹنہٕ، تہٕ اَتھ پؠٹھ راج کَرن والٮ۪ن آو «مُلوکُ الطوائف»، یعنی اَلگ اَلگ سلطنتَن ہٕندۍ بادشاہ، وَننہٕ۔ یہِ دَور روٗد ۲۳۰ عیسوی تام۔ اَتھۍ دَورَس منٛز آو عراق کٮ۪ن کینٛژن علاقَن پؠٹھ قحطٲنیَن ہُنٛد قبضہٕ، تہٕ پَتہٕ آیہِ کینہہ عدنٲنی تہِ، یِمَن ہٕنٛز تِمَن سۭتۍ بَین النَہرَین (Mesopotamia) کٮ۪ن کینٛژن حِصّن منٛز شَرٲکت گٔیہِ۔
[صَفہٕ 15]

اَردشیرَن (Ardashir) ٲس ۲۲۶ عیسوی منٛز ساسانی سلطنت قٲیِم کٔرمٕژ۔ تَمہِ سٕنٛزِ قیادتَس منٛز گوٚو فارسی دُبارٕ اَکس تہٕ طاقَتوَر۔ تِمو آیہِ پَننہِ سلطنتٕکٮ۪ن نَزدیکی علاقَن منٛز روزن والۍ عَرَب پَننِس قابوٗس تَل اَننۍ، تہٕ کٔرِکھ قُضاعہ قبیلَس شامَس کُن گژھنہٕ پؠٹھ مَجبوٗر۔ حیرہ (Heerah) تہٕ اَنبار کۍ لوٗکھ روٗدۍ فارسی حُکمرٲنی تَل۔
[صَفہٕ 15]

اَردشیر کِس دَورَس منٛز اوس جُذَیمہ الوَضّاح (Juzaima al-Waddah) حیرہ، رَبیعہ، مُضَر تہٕ بَین النَہرَینَس پؠٹھ حُکمرٲنی کَران۔ اَردشیرَن ٲس سَمَجمٕژ زِ عَرَبَن پؠٹھ بَراہِ راست حُکمرٲنی کَرٕنۍ تہٕ تِمَن پَننٮ۪ن سَرحَدَن پؠٹھ حملہٕ کَرنہٕ نِش روٚکُن چھُ مُشکِل، یوت تام نہٕ تِمَن منٛزے کانٛہہ تِتھ شَخص بادشاہ بَناوِ، یَتھ پَننِس قبیلٕچ طاقَت تہٕ حمایت حٲصِل آسہِ۔ تَمس یہِ تہِ باسیو زِ تِمَن ہٕنٛد فٲیدٕ ہٮ۪کِہِ سُہ تِمَن بازَنطینی بادشاہن خلاف تُلِتھ، یِم تَمس ہمیشہٕ پریشان کَران ٲسۍ۔ اَکہِ وَقتہٕ ہٮ۪کۍ عراق کۍ عَرَب شام کٮ۪ن عَرَبَن ہُنٛد مُقابلہٕ تہِ کٔرِتھ، یِم بازَنطینی بادشاہن ہٕندِس قابوٗس تَل ٲسۍ۔ مگر اَردشیرَن سَمَجھ زِ حیرہ کِس بادشاہ سٕندِس حُکمَس تَل گژھِ اَکھ فارسی فوجی دستہٕ تہِ آسُن، یُتھ سُہ تِمَن عَرَبَن خلاف اَتھہٕ یِیہِ، یِم تَمہِ خلاف بغاوت کَرن۔
[صَفہٕ 15–16]

جُذَیمہ تقریباً ۲۶۸ عیسوی منٛز فوت گژھنہٕ پَتہٕ، بَنٲو فارسی بادشاہ شاپوٗر بن اَردشیرَن عَمرو بن عَدی بن نَصر اللّخمی بادشاہ۔ عَمرو اوس لَخمی بادشاہن منٛز گوڈنیک، یِمَن حیرہَس پؠٹھ راج کوٚر۔ یہِ سِلسِلہٕ روٗد تیوٚت تام یوت تام فارسیَن قُباد بن فَیروزَس اِقتِدار دیُت۔ تَمہِ سٕندِس دَورَس منٛز آو مَزدَک (Mazdak) ناوٕ شَخص، یٔمۍ سَمٲجی زِندگی ہٕندٮ۪ن قائِدَن ہٕنٛز پٲبَندی ہَٹاونس ہٕنٛز دَعوت دِژ؛ کِتابہِ منٛز چھُ اَتھ بے راہ رَوی وَننہٕ آمُت۔ قُبادَن تہٕ تَمہِ سٕنٛزِ رَعایا ہٕندِس بٔڑِس حِصّس قبول کوٚر مَزدَکُک مذہب۔ تِمو حیرہ کِس بادشاہ مُنذِر بن ماء السّماءَس تہِ اَتھ پؠٹھ پَکنُک حُکم دیُت۔ ییلہِ تٔمۍ پَننہِ عزّت تہٕ خۄدّاری کِنۍ یہِ حُکم ماننہٕ نِش اِنکار کوٚر، قُبادَن ہَٹوو سُہ تہٕ تَمہِ جایہِ حارِث بن عَمرو بن حَجَر الکِندی مُقَرّر کوٚر، یٔمۍ مَزدَک سُنٛد نظریہٕ قبول کوٚرمُت اوس۔
[صَفہٕ 16]

قُباد پَتہٕ ییلہِ کِسریٰ اَنو شِیروان اِقتِدارَس پؠٹھ آو، تٔمۍ مَزدَک کِس نظریَس نِش نَفرَت آسنہٕ کِنۍ مور مَزدَک تہٕ تَمہِ سٕندۍ واریاہ پیروکار۔ تٔمۍ بَحال کوٚر مُنذِر حیرہ کِس تَختَس پؠٹھ تہٕ دیُت حارِثَس گرِفتار کَرنُک حُکم۔ حارِثَن نِیۍ اَلکَلب قبیلَس نِش پَناہ تہٕ روزٕے پَنٕنۍ باقی زِندگی تَتیٚتھ۔
[صَفہٕ 16]

مُنذِر بن ماء السّماء سٕندۍ نِچو روٗدۍ واریاہ کال بادشٲہی کَران، یوت تام نُعمان بن المُنذِرَس اِقتِدار میول۔ زَید بن عَدی العَبّادی سٕنٛزِ اَکہِ اَپوٗز اِلزامہٕ کِنۍ گوٚو فارسی بادشاہ نُعمانَس سۭتۍ ناراض تہٕ بُلاوُن تَمس پَننِس مَحلَس۔ نُعمان گوٚو پوٗشیدٕ پٲٹھۍ شَیبان قبیلٕکِس رَہنما ہانی بن مَسعودَس نِش، تہٕ تَمہِ حِفاظتَس منٛز تھاوُن پَنُن خاندان تہٕ مال۔ پَتہٕ گوٚو سُہ فارسی بادشاہ سٕندِس دَربارَس منٛز حٲضِر۔ بادشاہن ترٲو سُہ فوراً قَیدَس منٛز، ییتہِ تَمہِ ہٕنٛز موت گٔیہِ۔ پَتہٕ بَنٲو کِسریَٰن اِیاس بن قَبیصہ الطّائی حیرہُک بادشاہ۔ اِیاسَس آو حُکم دِنہٕ زِ ہانی بن مَسعودَس وَنِ نُعمان سٕنٛز اَمانت، یعنی خاندان تہٕ مال، کِسریَٰس حَوالہٕ کَرِ۔ عَرَب رَہنماہَن دیُت پَکّہٕ اِنکار۔ یہِ جواب میلنہٕ سۭتۍ کوٚر فارسی بادشاہَن شَیبان قبیلَس خلاف جَنٛگُک اِعلان، تہٕ اِیاس سٕنٛزِ قیادتَس منٛز سوٗزُن پَنُن لشکر ذی قارَس (Dhee Qar) کُن۔ تَتیٚتھ گوٚو زبردست جَنٛگ، یَتھ منٛز عَرَبَن گوڈنیک لَٹہِ تٲریخَس منٛز فارسیَن سخت شکست دِژ۔ کِتابہِ ہٕندِس بیانَس مُطٲبِق اوس یہِ نبی محمد ﷺ زاینہٕ تھوٚڑٕ کال پَتہٕ، تہٕ اِیاس حیرہ کِس تَختَس پؠٹھ بَہنہٕ اَٹھ رٮ۪تہٕ پَتہٕ گوٚمُت۔
[صَفہٕ 16]

اِیاس پَتہٕ آو حیرہَس پؠٹھ اَکھ فارسی حُکمران مُقَرّر کَرنہٕ، مگر ۶۳۲ عیسوی منٛز ییلہِ المُنذِر المَعرورَن اِقتِدار نیو، حُکمرٲنی گٔیہِ لَخم سٕندِس خاندانَس کُن واپس۔ تَمہِ حُکمرٲنی ہٕندۍ اوسۍ اَٹھ رٮ۪تہٕ تہِ مُشکِل سَپدِتھ، ییلہِ خالد بن الولیدَن مُسلمان سِپاہی ہٮ۪تھ تَمس پؠٹھ حملہٕ کوٚر۔ [محاضرات تاریخ الامم الاسلامیۃ 1/29–32]
[صَفہٕ 16]

شامٕچ حُکمرٲنی

قبیلَن ہٕنٛزِ ہجرت دَوران وٲتۍ قُضاعہ سٕندۍ کینہہ بجہ شام کٮ۪ن سَرحَدَن تام، تہٕ گوٚو تَتیٚتھ آباد۔ یِم ٲسۍ سُلَیح بن حُلوان سٕندِ خاندانُک حِصّہٕ۔ اَمہِ سٕندِ اولاد منٛز ٲسۍ دُجعَم بن سُلَیح سٕندۍ نِچو، یِمَن ضَجاعِمہ وَنان ٲسۍ۔ بازَنطینی ٲسۍ یِمَن قُضاعہ والٮ۪ن ہٕنٛز مَدَتھ عَرَب بَدوی حملہٕ آورَن تہٕ فارسیَن خلاف پَننٮ۪ن سَرحَدَن ہٕنٛزِ حِفاظت خٲطرٕ نِوان۔ یِمَن روٗد واریاہ کال خۄدمُختٲری حٲصِل؛ وَننہٕ چھُ یِوان زِ پوٗرِ دوٚیمہِ عیسوی صَدی ہُنٛد دَور اوس اَمہِ منٛز شٲمِل۔ زِیاد بن ہَبولہ اوس تِمَن ہٕندٮ۪ن مَشہوٗر بادشاہن منٛز اَکھ۔ مگر تِمَن ہُنٛد اِقتِدار گوٚو غَسّانیَن نِش شکست کھٮ۪نہٕ پَتہٕ ختم۔ پَتہٕ آو شام کٮ۪ن عَرَبَن پؠٹھ نُمٲیِندگی ہٕندِ طور حُکمرٲنی ہُنٛد اِختیار غَسّانیَن دِنہٕ، تہٕ دُومَۃُ الجَندَل (Dumat al-Jandal) بنیوو تِمَن ہُنٛد مَرکز۔ یہِ اِقتِدار روٗد ۱۳ ہجری کِس یَرموٗک (Yarmuk) جَنٛگَس تام۔ تِمَن ہٕندِ ٲخری بادشاہ جَبَلہ بن اَیہَمَن امیرُ المؤمنین عمر بن الخطاب رضی اللہ عنہ سٕندِس دَورَس منٛز اسلام قبول کوٚر۔ [محاضرات تاریخ الامم الاسلامیۃ 1/34؛ تاریخ ارض القرآن 2/80–82]
[صَفہٕ 16–17]

حجازٕچ حُکمرٲنی

اسماعیل علیہ السلامَس روٗد پَننہِ پوٗرِ زِندگی مکہٕچ حُکمرٲنی تہٕ بَیتُ اللہ ہٕنٛز نِگرٲنی پَننِس اَتھَس منٛز۔ ۱۳۷ ؤری ہٕنٛزِ عُمرِ منٛز تِمَن ہٕنٛزِ وَفات پَتہٕ آیہِ تِمَن ہٕندۍ زٕ نِچو، نابت تہٕ قیدار، تِمَن ہٕنٛزِ جایہِ۔ پَتہٕ نیو تِمَن ہٕندِ نانٮ۪ن، مُضاض بن عَمرو الجُرہُمیَن، اِقتِدار؛ اَمہِ پٲٹھۍ گٔیہِ مکہٕچ حُکمرٲنی جُرہُم قبیلَس کُن۔ اسماعیل سٕندٮ۪ن نِچوَن روٗد بَیتُ اللہ تعمیٖر کَرنَس منٛز پَننِس مالِ سٕنٛزِ خِدمت کِنۍ عزّت وول دَرجہٕ، مگر اِختیار اوس واریاہ کم۔ تِمَن ہٕنٛز یہِ حیثیت روٗد تیوٚت تام بَرقَرار، یوت تام بُخت نَصَرَس (Nebuchadnezzar) اُبھرنہٕ تھوٚڑٕ کال برونٛہہ جُرہُم قبیلُک زوال شروٗع گوٚو۔ [ابن ہشام 1/111]
[صَفہٕ 17]

مکہَس منٛز عدنٲنیَن ہُنٛد سیاسی حِصّہٕ اوس مَضبوٗط گژھان۔ اَمیُک صاف سُبوٗت چھُ یہِ زِ ییلہِ بُخت نَصَرَن ذاتِ عِرقَس منٛز عَرَبَن پؠٹھ گوڈنیک حملہٕ کوٚر، تَتھ وِزِ اوس عَرَبَن ہُنٛد رَہنما جُرہُم قبیلُک نہٕ۔ [قلب جزیرۃ العرب، صَفہٕ 230]
[صَفہٕ 17]

مگر ۵۸۷ قبل مسیح منٛز بُخت نَصَر سٕندِ دوٚیمِس حملَس کِنۍ دوٚڑۍ عدنٲنی یمنَس کُن۔ بَرمِیا نَبی گوٚو مَعَدّ ہٮ۪تھ شامَس کُن۔ ییلہِ بُخت نَصَر سُنٛد دَباو کم گوٚو، مَعَدّ آو مکہ واپس۔ تَتھ وِزِ میٚلِس نہٕ جُرہُم قبیلُک کانٛہہ نَفَر، جُرشَم بن جَلہَمہ علاوٕ۔ تَمہِ سٕنٛز کوٗر مُعانہ آیہِ مَعَدّ سۭتۍ نِکاحَس منٛز دِنہٕ، تہٕ پَتہٕ زاو تَمس تَمہِ نِش نِزار ناوٕ نِچو۔ [رحمۃ للعالمین 2/48]
[صَفہٕ 17]

مکہَس منٛز زِندگی ہٕندٮ۪ن سخت حالاتن تہٕ غریبی کِنۍ لوگۍ جُرہُم والۍ بَیتُ اللہ ہٕندٮ۪ن مِہمانَن سۭتۍ بُرا سُلوٗک کَران تہٕ اَمِچ دولت ناحَق اَتھہٕ کَران۔ اَمہِ سۭتۍ آو عدنٲنیَن، یعنی بَکر بن عبدِ مَناف بن کِنانہ سٕندِ اولادَس، غُصّہٕ تہٕ نَفرَت۔ مَرّ الظَہران ناوٕ ہَمسایہٕ علاقَس منٛز آمِتِس خُزاعہ قبیلٕچ مَدَتھ نِتھ، کٔرِکھ جُرہُمَس پؠٹھ حملہٕ تہٕ کَڑِکھ تِم مکہَس منٛزٕ نٮ۪بر۔ اَتھ جُملَس منٛز چھُ اَنٛگریزی ماخَذَس دوٚیمہِ عیسوی صَدی ہٕندِس مَنٛزِمَس دَورَس منٛز حُکمرٲنی «قُضاعہ» سٕندِس اَتھَس منٛز گژھنُک ذِکر؛ اَمہِ پَتہٕ چھُ متن خُزاعہ سٕنٛز حُکمرٲنی بیان کَران۔
[صَفہٕ 17]

مکہ ترٲونہٕ وِزِ بٔر جُرہُم والو زَمزَمُک کوٗہ، ہَموار کٔرِکھ اَمِچ جایہِ تہٕ دَفن کٔرِکھ اَتھ منٛز واریاہ چیٖزٕ۔ مَشہوٗر تٲریخ دان ابن اسحاق چھُ نَقل کَران زِ عَمرو بن الحارِث بن مُضاض الجُرہُمیَن کٔرۍ یمنَس کُن غَمگین حالتَس منٛز نیرنہٕ برونٛہہ زَمزَمَس منٛز زٕ سونُکۍ ہَرَن، حَجرِ اَسوَد، تہٕ واریاہ زیوَر تہٕ تلوارٕ دَفن۔ [ابن ہشام 1/114–115]
[صَفہٕ 17]

اسماعیل علیہ السلام سُنٛد دَور چھُ تقریباً ۲۰ صَدی قبل مسیح ہُنٛد ماننہٕ یِوان۔ اَمہِ حِسابہٕ سۭتۍ روٗد جُرہُم قبیلہٕ مکہَس منٛز تقریباً ۲۱ صَدی تہٕ کوٚرُن تقریباً ۲۰ صَدی تَتیٚتھ حُکمرٲنی۔
[صَفہٕ 17]

جُرہُمَس شکست میلنہٕ پَتہٕ آو مکہٕچ حُکمرٲنی صِرف خُزاعہ قبیلٕکِس اَتھَس منٛز۔ مگر مُضَر کٮ۪ن قبیلَن روٗدۍ ترٛٮ۪یہِ خٲص حَق حٲصِل:
[صَفہٕ 17]

گوڈنیک: حٲجِیَن عَرَفاتہٕ پؠٹھٕ مُزدَلِفَس تہٕ پَتہٕ مِنیٰ پؠٹھٕ عَقَبہ کِس جَمرَس تام نِنۍ۔ یہِ اِختیار اوس غَوث بن مُرّہ سٕندِس خاندانَس، یُس اِلیاس بن مُضَر سٕنٛز اَکھ بَج اوس۔ تِمَن اوس «صوفہ» وَننہٕ یِوان۔ اَمہِ حَقُک مطلب اوس زِ حٲجی ہٮ۪کۍ نہٕ عَقَبہ کِس جَمرَس کٲنٛہۍ ترٲوِتھ، یوت تام نہٕ صوفہ ہُنٛد کانٛہہ نَفَر گوڈٕ اَمہِ پٲٹھۍ کَرِ۔ کٲنٛہۍ ترٲوِتھ ییلہِ تِم مِنیٰ وادی ترٲوُن یَژھان ٲسۍ، صوفہ والۍ ٲسۍ عَقَبہ کٮ۪ن دۄشوَے پَکھَن وۄتھان۔ کانٛہہ ہیٚکِہِ نہٕ تَتہٕ پَکِتھ، یوت تام نہٕ صوفہ والۍ پَکۍ ہٮ۪تھ حٲجِیَن خٲطرٕ وَتھ خٲلی کَرن۔ ییلہِ صوفہ ختم گوٚو، یہِ ذِمّہٕ وٲری آیہِ تَمیم قبیلٕکِس سَعد بن زَید مَنات سٕندِس خاندانَس کُن۔
[صَفہٕ 17–18]

دوٚیم: قُربٲنی کِس دۄہَکِس صُبحَس مُزدَلِفہ پؠٹھٕ مِنیٰ کُن پَکنٕچ، یعنی اِفاضہٕ کَرناونٕچ، ذِمّہٕ وٲری۔ یہِ ٲس عَدوان سٕندِس خاندانَس۔
[صَفہٕ 18]

ترٛٮ۪یم: حُرمَت والٮ۪ن رٮ۪تَن ہٕنٛز ترتیٖب پَتھ کُن کَرٕنۍ، یعنی نَسیء۔ یہِ ذِمّہٕ وٲری ٲس بَنو کِنانہ ہٕندِس تَمیم بن عَدی سٕندِس خاندانَس۔
[صَفہٕ 18]

خُزاعہ سٕنٛز مکہَس منٛز حُکمرٲنی روٗد ۳۰۰ ؤری۔ اَتھ دَورَس منٛز پھیٚلۍ عدنٲنی سورِے نَجدَس منٛز، تہٕ بحرین تہٕ عراق کٮ۪ن پَکھَن کُن۔ قُریشَن ہٕندۍ لوکٕٹۍ بجہ روٗدۍ مکہ ہٕنٛدٮ۪ن اَندن پَکھَن؛ تِم ٲسۍ حُلول، حُروم تہٕ کِنانہ سٕندۍ کینہہ خاندان۔ تِمَن اوس نہٕ مکہَس یا بَیتُ اللہَس منٛز کانٛہہ خٲص حَق، یوت تام قُصَیّ بن کِلاب آو برونٛہہ۔ وَننہٕ چھُ یِوان زِ قُصَیّ سُنٛد مول گوٚو تَمہِ لوکُٹ شُر آسنہٕ وِزِ فوت۔ پَتہٕ گوٚو تَمہِ ہٕنٛزِ ماجہِ ہُنٛد نِکاح بَنو عُذرہ قبیلٕکِس رَبیعہ بن حَرامَس سۭتۍ۔ رَبیعہَن نِیۍ پَنٕنۍ زنانہٕ تہٕ تَمہِ سُنٛد شُر شام کٮ۪ن سَرحَدَن پؠٹھ پَننِس وَطَنَس۔ ییلہِ قُصَیّ جَوان گوٚو، سُہ آو مکہ واپس۔ تَتھ وِزِ اوس خُزاعہُک حُلَیل بن حَبشہ تَتیٚتھ حُکمران، یٔمۍ کٔر پَنٕنۍ کوٗر حُبّیٰ قُصَیّ سۭتۍ نِکاحَس منٛز۔ حُلَیل فوت گژھنہٕ پَتہٕ شروٗع گوٚو خُزاعہ تہٕ قُریشَن دَرمیان جَنٛگ، یَمہِ نتیٖجَس منٛز آو مکہ تہٕ بَیتُ اللہ قُصَیّ سٕندِس قابوٗس منٛز۔ [ابن ہشام 1/117]
[صَفہٕ 18]

جَنٛگٕکۍ ترٛٮ۪یہِ بیان

گوڈنیک بیان: حُلَیل فوت گژھنہٕ پَتہٕ وُچھ قُصَیَّن پَننِس اولادُک پھیٚلاو، پَننہِ دولتُک زیادٕ گژھُن تہٕ پَننہِ عزّتُک بَڑُن۔ تَمس باسیو زِ مکہٕچ حُکمرٲنی تہٕ بَیتُ اللہ ہٕنٛز نِگرٲنی ہٕنٛز ذِمّہٕ وٲری سَمبٲلنُک حَق چھُ خُزاعہ تہٕ بَنو بَکر کھۄتہٕ زیادہٕ تَمس۔ تٔمۍ یہِ تہِ ووٚن زِ قُریش چھِ اسماعیل سٕندِ اولادُک سَردار۔ تَمہِ پَتہٕ کٔر تٔمۍ قُریش تہٕ کِنانہ کٮ۪ن کینٛژن لوٗکھن سۭتۍ خُزاعہ تہٕ بَنو بَکر مکہَس منٛزٕ کَڑنہٕ بارَس منٛز مَشوَرٕ۔ تِمَن گوٚو اَمیُک خیال پسند تہٕ دِژِکھ تَمس حمایت۔
[صَفہٕ 18]

دوٚیم بیان: خُزاعہ ہٕنٛدِس دعوَس مُطٲبِق اوس حُلَیلَن قُصَیّ سِفارش کٔرمُت زِ تَمہِ وَفات پَتہٕ سَمبٲلِ سُہ کعبہٕ شَریفٕچ نِگرٲنی تہٕ مکہٕچ حُکمرٲنی۔
[صَفہٕ 18]

ترٛٮ۪یم بیان: حُلَیلَن دیُت کعبہٕ شَریفٕچ خِدمتُک حَق پَننہِ کوٗرِ حُبّیٰ ہٕنٛدِس اَتھَس منٛز، تہٕ مُقَرّر کوٚرُن ابو غَبشان الخُزاعی اَمہِ کامہٕ خٲطرٕ تَمہِ سُنٛد وَکیٖل۔ حُلَیل فوت گژھنہٕ پَتہٕ ہیوٚت قُصَیَّن یہِ حَق چَرمہٕ کِس اَکِس شَرابہٕ تھیٚلَس بدلہٕ۔ اَمہِ سۭتۍ گوٚو خُزاعہ والۍ ناراض تہٕ کوٚرُکھ کعبہٕ شَریفٕچ نِگرٲنی قُصَیَّس نِش الگ تھاونٕچ کوٗشِش۔ مگر قُصَیّ گوٚو قُریش تہٕ کِنانہ ہٕنٛزِ مَدَتہٕ سۭتۍ حُکمرٲنی حٲصِل کَرنَس منٛز کامیاب، تہٕ کَڑُن خُزاعہ پوٗرٕ پٲٹھۍ مکہَس منٛزٕ نٮ۪بر۔ [رحمۃ للعالمین 2/55]
[صَفہٕ 18]

پوٗز کیاہ تہِ آسہِ، اَمہِ سورِے مُعاملہٕ ہُنٛد نتیٖجہٕ گوٚو یہِ زِ صوفہ گوٚو برونٛہہ بیان کَرنہٕ آمٕتٮ۪ن حَقَن نِش مَحروٗم، خُزاعہ تہٕ بَکر آیہِ مکہَس منٛزٕ کَڑنہٕ، تہٕ مکہٕچ حُکمرٲنی تہٕ بَیتُ اللہ ہٕنٛز نِگرٲنی آیہِ قُصَیَّس کُن۔ اَمہِ برونٛہہ ٲسۍ قُصَیّ تہٕ خُزاعہ دَرمیان سخت جَنٛگ گوٚمٕتۍ، یِمَن منٛز دۄشوَنی پَکھَن واریاہ جانٕکۍ نُقصان گوٚو۔ پَتہٕ گٔیہِ صُلح، تہٕ بَکر قبیلٕکِس یَعمُر بن عَوفَس آو فیصلہٕ کَرنہٕ خٲطرٕ ماننہٕ۔ تَمہِ سٕندِس فیصلَس مُطٲبِق اوس قُصَیّ مکہٕچ حُکمرٲنی تہٕ بَیتُ اللہ ہٕنٛز نِگرٲنی ہُنٛد حَق دار؛ خُزاعہ سٕنٛزِ خون ریزی ہٕنٛز ذِمّہٕ وٲری ٲس نہٕ تَمس پؠٹھ، تہٕ دِیَت دِنُک حُکم آو خُزاعہَس دِنہٕ۔ قُصَیّ سٕنٛز مکہ تہٕ بَیتُ اللہَس پؠٹھ حُکمرٲنی شروٗع گٔیہِ ۴۴۰ عیسوی منٛز۔ اَمہِ سۭتۍ آو تَمس، تہٕ پَتہٕ قُریشَن، مکہَس پؠٹھ پوٗرٕ اِقتِدار تہٕ بَیتُ اللہ ہٕنٛزِ نِگرٲنی ہُنٛد بے مُقابلہٕ حَق حٲصِل؛ ساروٕے عَرَبہٕ پؠٹھٕ ٲسۍ لوٗکھ تَتھ مُقدّس گَرَس کُن یِوان۔
[صَفہٕ 18–19]

مکہ تہٕ اَمِکۍ اِدارٕ

قُصَیَّن آنۍ پَنٕنۍ رِشتٕ دار مکہَس منٛز تہٕ بانٹُن اَمیٕکۍ علاقہٕ تِمَن منٛز۔ تٔمۍ دِژ قُریشَن تہِ تَتیٚتھ روزنٕچ جایہِ۔ اَلنُّساۃ، صَفوان، عَدوان تہٕ مُرّہ بن عَوف سٕندٮ۪ن خاندانَن روٗدۍ تِمے حَق حٲصِل، یِم تِمَن قُصَیّ یِنہٕ برونٛہہ ٲسۍ۔ [ابن ہشام 1/124]
[صَفہٕ 19]

قُصَیّ سٕنٛز اَکھ اہم کامیٲبی ٲس کعبہٕ شَریف ہٕنٛدِ شُمالی پَکھَس پؠٹھ دارُ النَّدوہ (Dar an-Nadwa)، یعنی مَشوَرٕ کَرنہٕ خٲطرٕ جمع گژھنُک گَرٕ، قٲیِم کَرُن۔ یہِ اوس قُریشَن ہٕنٛزَن مَجلسَن ہُنٛد مَرکز۔ اَمہِ گَرَن کٔر قُریشَن واریاہ مَدَتھ، تکیازِ تِمَن ہٕنٛز رائے ٲس اَکس کَران تہٕ تِمَن ہٕندۍ مَسلہٕ ٲس باہَمی مِہربٲنی سۭتۍ حَل کَران۔ [ابن ہشام 1/125؛ اخبار الکرام، صَفہٕ 152]
[صَفہٕ 19]

قُصَیَّس ٲسۍ رَہنُمٲیی تہٕ عزّتُکۍ یِم خٲص حَق حٲصِل:
[صَفہٕ 19]

۱۔ دارُ النَّدوہ ہٕنٛزَن مَجلسَن ہٕنٛز صَدارَت، یِمَن منٛز بٔڑٮ۪ن مُعاملن پؠٹھ مَشوَرٕ ٲس سَپدان تہٕ نِکاحَن ہُنٛد اِعلان ٲس کَرنہٕ یِوان۔
۲۔ جھنڈٕ: جَنٛگ شروٗع کَرنَس سۭتۍ جُڑمٕتۍ مُعاملہٕ ٲسۍ صِرف تَمہِ سٕندِس اِختیارَس منٛز۔
۳۔ کعبہٕ شَریفٕچ دَروازٕ بانی: اَمیُک دَروازٕ کھولنُک حَق اوس صِرف تَمس، تہٕ خِدمت تہٕ حِفاظتٕچ ذِمّہٕ وٲری تہِ ٲس تَمس۔
۴۔ حٲجِیَن آب دِیُن: سُہ اوس تِمَن خٲطرٕ آبُکۍ حَوض بَراوان، یِم کھٲژرَن تہٕ دَچھَن سۭتۍ مِٹھۍ کَرنہٕ ٲسۍ یِوان۔
۵۔ حٲجِیَن کھٮ۪ن دِیُن: یِمَن پَنُن کھٮ۪ن سَمبٲلنُک وُس اوس نہٕ، تِمَن خٲطرٕ کھٮ۪ن تیٲر کَراوُن۔ قُصَیَّن تھاو قُریشَن پؠٹھ اَمہِ کھٮ۪نہٕ خٲطرٕ سالانہٕ لگان تہِ، یُس حَج کِس موسِمَس منٛز اَدا کَرنہٕ ٲس یِوان۔ [ابن ہشام 1/130]
[صَفہٕ 19]

اَتھ کتھہِ پؠٹھ گژھِ غور کَرُن زِ قُصَیَّن کٔر پَننِس نِچوَس عبدِ مَنافَس خٲص عزّت تہٕ مَرتبہٕ دِنہٕ، حالانٛکہِ سُہ اوس نہٕ تَمہِ سُنٛد بٔڈ نِچو؛ بٔڈ نِچو اوس عبدُ الدّار۔ اَتھ فراہم کَرنہٕ آمِتِس اَنٛگریزی متنَس منٛز چھُ دارُ النَّدوہ ہٕنٛز صَدارَت، جھنڈٕ، کعبہٕ شَریفٕچ دَروازٕ بانی، تہٕ حٲجِیَن آب تہٕ کھٮ۪ن دِنٕچ ذِمّہٕ وٲری عبدِ مَنافَس حَوالہٕ کَرنُک ذِکر۔ قُصَیّ سٕندٮ۪ن کٲمَن پؠٹھ ٲس نہٕ کانٛہہ سَوال تُلان تہٕ تَمہِ سٕندۍ حُکم ٲسۍ اَٹل ماننہٕ یِوان۔ اَمہِ کِنۍ گوٚو نہٕ تَمہِ وَفات پَتہٕ تَمہِ سٕندٮ۪ن نِچوَن منٛز جھگڑٕ۔ مگر پَتہٕ گوٚو پوتَرَن منٛز اِختِلاف۔ عبدِ مَناف فوت گژھنہٕ سۭتۍ لوگۍ تَمہِ سٕندۍ نِچو پَننٮ۪ن پِتَر بٲیَن، یعنی عبدُ الدّار سٕندٮ۪ن نِچوَن، سۭتۍ جھگڑٕ کَران۔ اَمہِ کِنۍ ہیٚکِہِ ہا سورِے قُریش قبیلَس منٛز پھوٗٹ تہٕ جَنٛگ گٔژِتھ، اَگَر نہٕ اَکھ صُلح نامہٕ سَپُد ہا۔ تَمہِ مُطٲبِق آیہِ عُہدٕ دُبارٕ بانٹنہٕ: حٲجِیَن کھٮ۪ن تہٕ آب دِنٕچ ذِمّہٕ وٲری روٗد عبدِ مَناف سٕندٮ۪ن نِچوَن، تہٕ دارُ النَّدوہ، جھنڈٕ تہٕ کعبہٕ شَریفٕچ دَروازٕ بانی روٗد عبدُ الدّار سٕندٮ۪ن نِچوَن۔ عبدِ مَناف سٕندٮ۪ن نِچوَو ترٲو قُرعہٕ، تہٕ کھٮ۪ن تہٕ آب دِنُک کام آو ہاشِم بن عبدِ مَنافَس کُن۔ تَمہِ وَفات پَتہٕ آو یہِ کام تَمہِ سٕندِس بٲیِس مُطّلِب بن عبدِ مَنافَس، تہٕ پَتہٕ نبی کریم ﷺ سٕندِس دادَٕس عبدُالمُطّلِب بن ہاشِمَس کُن۔ تَمہِ سٕندۍ نِچو روٗدۍ اسلام یِنَس تام یہِ کام سَمبٲلان۔ اسلام یِنہٕ وِزِ ٲس یہِ ذِمّہٕ وٲری عباس بن عبدُالمُطّلِبَس۔ [ابن ہشام 1/129–179]
[صَفہٕ 19]

قُریشَن ہٕندٮ۪ن لوٗکھن منٛز آیہِ واریاہ باقی عُہدٕ تہِ بانٹنہٕ، یُتھ اَکہِ نَوِ لوکٕٹِس ریاستٕچ بُنیاد مَضبوٗط گژھِ۔ مُصنّف چھُ اَمِکٮ۪ن دَفتَرَن تہٕ مَجلسَن ہٕنٛز مُشابَہَت اَزکٮ۪ن حُکومتی اِدارَن سۭتۍ دِوان تہٕ اَتھ جَمہوٗری لوکٕٹ ریاست وَنان۔ یِمن عُہدَن منٛز کینہہ ٲسۍ یِم:
[صَفہٕ 20]

۱۔ بُتَن ہٕندِ ناوٕ فال کَڑنُک کام اوس بَنو جُمَحَس۔
۲۔ نَذرَن تہٕ قُربٲنیَن ہُنٛد حِساب تھاوُن، جھگڑٕ تہٕ اَتھ سۭتۍ جُڑمٕتۍ مُعاملہٕ حَل کَرُن اوس بَنو سَہم سٕندِس اَتھَس منٛز۔
۳۔ مَشوَرٕ دِنُک کام اوس بَنو اَسدَس۔
۴۔ دِیَت تہٕ جُرمانَن ہُنٛد اِنتِظام اوس بَنو تَیمَس۔
۵۔ قَومی جھنڈٕ تُلنٕچ ذِمّہٕ وٲری ٲس بَنو اُمیّہَس۔
۶۔ فوجی اِدارٕ، پَیدَل سِپاہی تہٕ سوار ٲسۍ بَنو مَخزوٗم سٕنٛز ذِمّہٕ وٲری۔
۷۔ بَنو عَدی ٲسۍ باقی قَومَن سۭتۍ سَفیٖری تہٕ دَرمیانی مُعاملہٕ سَمبٲلان۔ [تاریخ ارض القرآن 2/104–106]
[صَفہٕ 20]

سورِے عَرَبُک حال

اَسہِ چھُ برونٛہہ قحطٲنی تہٕ عدنٲنی ہجرتَن ہُنٛد ذِکر کوٚرمُت، تہٕ یہِ وَنمُت زِ عَرَب ہٕندۍ علاقہٕ کِتھہٕ پٲٹھۍ یِمن دۄن قَومَن منٛز بانٹنہٕ آیہِ۔ یِم قبیلہٕ حیرہ نِش روزان ٲسۍ، تِم ٲسۍ حیرہ کِس عَرَب بادشاہَس تَل۔ شام کِس اَدھ ریگستانی علاقَس منٛز روزن والۍ ٲسۍ عَرَب غَسّانی بادشاہ سٕندِس قابوٗس تَل۔ مگر یہِ تابعدٲری ٲس حَقیقَتَس منٛز زیادہٕ تَر صِرف ناوُک ہُنٛد، عَملَس منٛز نہٕ۔ اَندرِمٮ۪ن دُور ریگستانَن منٛز روزن والٮ۪ن اوس پوٗرٕ خۄدمُختٲری حٲصِل۔
[صَفہٕ 20]

یِمن قبیلَن ہٕندۍ رَہنما ٲسۍ سورِے قبیلَن چُنان۔ یہِ اوس اَکھ اَدھ حُکومت ہِش اِنتِظام، یُس قبٲیِلی اِتّحاد تہٕ زَمیٖن تہٕ مالٕچ رٲچھ کَرنہٕ خٲطرٕ مِلۍ جُلۍ مَفادَن پؠٹھ بَنٲوِتھ اوس۔
[صَفہٕ 20]

قبیلَن ہٕندٮ۪ن رَہنماہن ٲسۍ بادشاہن ہِشے بے روک اِختیار حٲصِل۔ جَنٛگ تہٕ اَمن دۄشوَے منٛز ٲس تِمَن ہٕنٛز پوٗرٕ فرمانبردٲری کَرنہٕ یِوان۔ مگر حُکمرٲنی خٲطرٕ پِتَر بٲیَن دَرمیان مُقابلہٕ ٲس واریاہ لَٹہِ تِمَن اَکھ اَکِس کھۄتہٕ برونٛہہ نیرنہٕ پؠٹھ مَجبوٗر کَران۔ تِم ٲسۍ مِہمان نَوازی، سَخاوت، دانٲیی تہٕ بہادُری ہاوان، یُتھ پَننٮ۪ن مُقابِلَن کھۄتہٕ بَرتر بَنن تہٕ لوٗکھن، خاص پٲٹھۍ شاعِرَن، منٛز ناو پٲدٕ کَرن۔ شاعِر ٲسۍ تَمہِ وَقتہٕ قَومٕکۍ باقاعِدٕ تَرجُمان۔
[صَفہٕ 20]

قبیلَن ہٕندٮ۪ن رَہنماہن تہٕ سَردارَن اوس مالِ غنیمتَس منٛز خٲص حَق۔ تِمَن اوس غنیمتُک ژورِم حِصّہٕ میلان، تہٕ سۄ تہِ چیٖز یۄس تِم پانَس خٲطرٕ چُنن، یا واپسی ہٕنٛزِ وَتہِ پؠٹھ لَبن۔ بانٹنہٕ پَتہٕ باقی روٗد مال، یُس بیٚیہِ بانٹُن مُمکِن آسہِ نہٕ، سُہ تہِ اوس تِمَن ہُنٛد۔
[صَفہٕ 20]

سیاسی حالت

نیبرِمین سۭتۍ سَرحَد تھاون والۍ ترٛٮ۪یہِ عَرَب علاقہٕ ٲسۍ سخت کمزوٗری تہٕ پَسماندگی ہُنٛد شِکار۔ تَتِکۍ لوٗکھ ٲسۍ یا آقا یا غُلام، یا حُکمران یا مَحکوٗم۔ آقاَن، خاص پٲٹھۍ نیبرِمین، اوس ہر فٲیدَس پؠٹھ حَق، مگر غُلامَن پؠٹھ ٲسہِ صِرف ذِمّہٕ وٲری۔ یعنی بے لَگام حُکمرٲنی کِنۍ ٲسۍ مَحکوٗمَن ہٕندۍ حَق پامال گژھان، تہٕ تِمَن پؠٹھ ٲس جَہالَت، ظُلم، بے اِنصافی تہٕ سختی۔ تِم ٲسۍ تاریکی تہٕ بے زانی منٛز بَٹکان۔ تِمَن ہٕنٛز حالت ٲس زَرخیز زَمیٖنہِ ہِش، یَمہِ ہُنٛد پَیداوٲر حُکمران تہٕ طاقَتوَر لوٗکھ پانَس نِوان، تہٕ پَننہِ عَیش، خوشی، خواہِش، زوٗر آوَری تہٕ زیادتی پؠٹھ بے دَریٖغ خرچ کَران۔ یِمن علاقَن نِش روزن والۍ قبیلہٕ ٲسۍ کَتھ ساتہٕ شامَس کُن تہٕ کَتھ ساتہٕ عراقَس کُن رُخ کَران۔ عَرَب کٮ۪ن اَندرِمٮ۪ن علاقَن منٛز روزن والۍ ٲسۍ آپَس منٛز جُدا، تہٕ قبٲیِلی جَنٛگَن، نَسلی تہٕ مذہبی جھگڑَن ہٕندِس قابوٗس تَل۔
[صَفہٕ 20–21]

تِمَن اوس نہٕ کانٛہہ تِتھ بادشاہ یُس تِمَن ہٕنٛز آزٲدی تھاوِ ہا بَرقَرار، تہٕ نہٕ کانٛہہ تِتھ مَدَتھ کَرن وول، یَمہِ نِش تِم مُشکِل وَقتَس منٛز مَشوَرٕ ہٮ۪کَن ہا نِتھ یا یَتھ پؠٹھ بَروسہٕ ہٮ۪کَن ہا کٔرِتھ۔
[صَفہٕ 21]

حجاز کٮ۪ن حُکمرانَن ٲس البتہٕ عَرَبَن منٛز بٔڈ عزّت تہٕ قَدر۔ تِم ٲسۍ مذہبی مَرکزُکۍ حُکمران تہٕ خادِم ماننہٕ یِوان۔ حجازٕچ حُکمرٲنی ٲس دُنیٲوی تہٕ رَسمی بَرتری، تہٕ مذہبی رَہنُمٲیی ہُنٛد اَکھ مِلاوَٹ۔ تِم ٲسۍ مذہبی رَہنُمٲیی ہٕندِ ناوٕ عَرَبَن پؠٹھ حُکمرٲنی کَران، تہٕ بَیتُ اللہ تہٕ اَمہِ زوٚپٲر کِس علاقُک نِگرٲنی ہُنٛد حَق ٲسۍ ہمیشہٕ پَننِس اَتھَس منٛز تھاوان۔ تِم ٲسۍ کعبہٕ شَریف کٮ۪ن مِہمانَن ہُنٛد خیال تھاوان، تہٕ ابراہیم علیہ السلام سٕندٮ۪ن قائِدَن پؠٹھ عَمل کَرناونٕچ ذِمّہٕ وٲری ٲس تِمَن۔ تِمَن ہٕندۍ ٲسۍ اَزکٮ۪ن پارلیمانَن ہِشۍ دَفتَر تہٕ شُعبہٕ تہِ۔ مگر یہِ بۄژ سَمبٲلنہٕ خٲطرٕ ٲسۍ تِم واریاہ کمزوٗر، یَتھہٕ پٲٹھۍ حَبَشی حملہٕ وِزِ صاف برونٛہہ آو۔
[صَفہٕ 21]

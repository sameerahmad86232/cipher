# صحیح بخاری — حدیث 1 تا 10

## حدیث 1 — اعمال نیتن پؠٹھ دارومدار تھاوان

**راوی:** حضرت عمر بِن الخطاب رضی اللہ عنہ  
**حیثیت:** پہلے منظور شدہ

مےٚ بوٗز رَسوٗلُ اللہ ﷺ فَرموو: ”عملَن ہُنٛد اَجر چھُ نِیَتَن پؠٹھ دارومدار تھاوان، تہٕ ہر کانٛسہِ اِنسانَس مِلہِ سُے یِمِچ تٔمۍ نِیَت کٔرمٕژ آسہِ۔ پَس یِمۍ دُنیاوی فٲیدٕ حٲصِل کَرنہٕ خٲطرٕ، یا کَنہِ زَنانہِ سۭتۍ نِکاح کَرنہٕ خٲطرٕ ہِجرَت کٔر، تٔمۍ سٕنٛز ہِجرَت چھِ تَمَے خٲطرٕ، یَمَے خٲطرٕ تٔمۍ ہِجرَت کٔر۔“

## حدیث 2 — وَحی کِتھ کٔنۍ چھِ نازِل سَپدان؟

**راوی:** اُمّ المؤمنین حضرت عائشہ رضی اللہ عنہا  
**حیثیت:** پہلے منظور شدہ

حارث بِن ہشامَن پٔرژھ رَسوٗلُ اللہ ﷺ: ”یا رَسوٗلَ اللہ ﷺ! توہہ پؠٹھ کِتھ کٔنۍ چھِ وَحی نازِل سَپدان؟“ رَسوٗلُ اللہ ﷺ فَرموو: ”کُنہِ ساتہٕ چھِ مےٚ وَحی گَنٛٹہِ ہِنٛز آواز پٲٹھۍ یِوان۔ وَحیُک یِہ طریقہٕ چھُ مےٚ خٲطرٕ سٲریو کھۄتہٕ سَخت۔ ییلہِ یِہ وَحی مؠنِس دِلَس منٛز مَحفوٗظ چھِ گَژھان، تیلہِ چھِ یِہ حالَت خَتم گَژھان۔ کُنہِ ساتہٕ چھُ فِرِشتہٕ اَکہِ اِنسان سُنٛد شکلہٕ منٛز مےٚ نِش یِوان تہٕ مےٚ سۭتۍ کَتھ کَران؛ تہٕ یُس کینٛہہ سُہ چھُ وَنان، سُہ چھُ مؠنِس دِلَس منٛز مَحفوٗظ گَژھان۔“

حضرت عائشہ رضی اللہ عنہا چھِ مَزید روایت کَران: ”بے شَک، مےٚ وُچھ نَبی کریم صلی اللہ علیہ وسلم پؠٹھ اَکِس سَخت سَرد دۄہَس منٛز وَحی نازِل سَپدان۔ ییلہِ وَحی نازِل سَپٕد، مےٚ وُچھ نَبی کریم صلی اللہ علیہ وسلم سُنٛدِس ڑٮ۪کَس پؠٹھ عٲرَق وَسان۔“

## حدیث 3 — وَحی ہُنٛد شروٗعات

**راوی:** اُمّ المؤمنین حضرت عائشہ رضی اللہ عنہا  
**حیثیت:** پہلے جملہ بہ جملہ منظور شدہ

رَسوٗلُ اللہ صلی اللہ علیہ وسلمَن پؠٹھ وَحی نازِل سَپُد نیک تہٕ پوٚزَن خوابَن سۭتۍ شروٗع۔ نَبی کریم صلی اللہ علیہ وسلم یُس تہِ خواب اوس وُچھان، سُہ اوس صُبحُک گاشُک پٲٹھۍ صاف تہٕ پوٚز آسان۔ پَتہٕ آیہِ نَبی کریم صلی اللہ علیہ وسلمَن خَلوَت پَسند تہٕ تِم ٲس غارِ حرا ہَس منٛز کُنہِ زون آسان تہٕ ٲس اللہ تعالیٰ سٕنز عبادت کَران۔ تِم ٲس کَیَن کَیَن دۄہَن غارِ حرا ہَس منٛز روزان تہٕ اللہ تعالیٰ سٕنز عبادت کَران، یوتام تِمَن گَرٕکؠن وُچھنُک شوق اوس یِوان۔ تِم ٲس پانَس سۭتۍ کھؠن نِوان۔ پَتہٕ ٲس تِم حَضرت خدیجہ رضی اللہ عنہاؠس نِش واپس یِوان، تہٕ بَیِہ تَتَے پٲٹھۍ کھؠن نِتھ غارِ حرا ہَس کُن گَژھان۔ یہِ سِلسِلہٕ روٗد جاری یوتام حق سَپُد۔

فِرِشتہٕ آو تہٕ تٔمۍ ووٚن: ”پَرٕ!“ نَبی کریم صلی اللہ علیہ وسلمَن فَرموو: ”بہٕ چھُس نہٕ پَرُن زانان۔“ فِرِشتَن روٚٹُس بہٕ پانَس سۭتۍ زور سان۔ پَتہٕ ترٛووُس بہٕ تٔمۍ تہٕ ووٚنُن بَیِہ: ”پَرٕ!“ پَتہٕ روٚٹُس بہٕ تٔمۍ بَیِہ دۄیمہِ لَٹہِ پانَس سۭتۍ زور سان۔ پَتہٕ ترٛووُس بہٕ تٔمۍ تہٕ ووٚنُن بَیِہ: ”پَرٕ!“ مےٚ ووٚن: ”بہٕ چھُس نہٕ پَرُن زانان؛ بہٕ کیا پَرٕ؟“ پَتہٕ روٚٹُس بہٕ تٔمۍ ترٛیِمِ لَٹہِ پانَس سۭتۍ زور سان۔ پَتہٕ ترٛووُس بہٕ تٔمۍ تہٕ ووٚنُن: ”پَرٕ پَننِس رَبَّس سُنٛدِس ناوَس سۭتۍ، یِمۍ ساروے کینٛہہ پَیدا کوٚر۔ اِنسان کوٚر لَگِتھ کَمَتۍ خوٗنہٕ سۭتۍ پَیدٕ۔ پَرٕ، تہٕ تُہُنٛد رَب چھُ سٲریو کھۄتہٕ زیادٕ کَرَم کرن وول۔“

وَحی نازِل گَژِتھ کٔر نَبی کریم صلی اللہ علیہ وسلمَن گَرٕ کُن تَشریٖف۔ نَبی کریم صلی اللہ علیہ وسلم سُنٛد دِل اوس زور زور سان دَڑکان۔ نَبی کریم صلی اللہ علیہ وسلمَن فَرموو: ”مےٚ وَلِیو! مےٚ وَلِیو!“ حَضرت خدیجہ رضی اللہ عنہاؠن وول نَبی کریم صلی اللہ علیہ وسلم کَملی منٛز، یوتام نَبی کریم صلی اللہ علیہ وسلم سُنٛد خوف دور گوٚو۔ پَتہٕ فَرموو نَبی کریم صلی اللہ علیہ وسلمَن حَضرت خدیجہ رضی اللہ عنہاؠس سورُے سُہ یُس واقعہٕ سَپُدمُت اوس تہٕ فَرموو: ”مےٚ چھُ پَننِس پانَس مُتعلِق خَطرٕ مَحسوٗس سَپدان۔“

حَضرت خدیجہ رضی اللہ عنہاؠن ووٚن: ”زانہہ نہٕ! خُدایہِ سُنٛد قَسَم، اللہ تعالیٰ کَرِو نہٕ توہہِ زانہہ تہِ رُسوا۔ توہہ چھِ رِشتہٕ دارَن سۭتۍ اَصٕل تَعلُق تھاوان، کمزورَن ہُنٛد بار چھِ تُلان، غریٖبَن تہٕ محتاجَن چھِ مَدَد کَران، پازؠن چھِ عِزت کَران تہٕ مُصیٖبتَس منٛز پھَسِمَتؠن چھِ سہارا دِوان۔“

پَتہٕ نِیہِ حَضرت خدیجہ رضی اللہ عنہاؠن نَبی کریم صلی اللہ علیہ وسلم وَرَقہٕ بِن نَوفَلَس نِش۔ سُہ اوس اِنجل لیٚکھان عِبرٲنی منٛز، یوتاہ اللہ تعالیٰن مَنظوٗر اوسُس کوٚرمُت۔ سُہ اوس بوٚڈ گومُت تہٕ تَمس اوس نہٕ اَتھ تہِ یِوان۔ حَضرت خدیجہ رضی اللہ عنہاؠن ووٚن ورقہَس زِ: ”پَنٕنۍ بابتھر سٕنٛز کَتھ بوٗزِو، میانہِ پِتٕرۍ!“ ورقہَن پرٛژھُس: ”میون بابتھر! ژیہِ کیا وُچھمُت؟“ رسول اللہ صلی اللہ علیہ وسلمَن کوٚر سُہ سورُے بیان، یُس تِمو وُچھمُت اوس۔

ورقہَن ووٚن: ”یہِ چھُ سُے ناموٗس—فِرِشتہٕ جبرئیل علیہ السلام—یِمِس اللہ تعالیٰن حَضرت موٗسیٰ علیہ السلامَس نِش سوزمُت اوس۔ کاش بہٕ تَمہِ وِزِ جوان آسہٕ! کاش بہٕ گوٚوسُس تَتھ دۄہَس تام زِندٕ روزُن، ییلہِ تُہُنٛد قوم توہہِ ییٚتہِ کَڑنؠو!“ رسول اللہ صلی اللہ علیہ وسلمَن فَرموو: ”کیا تِم کَڑَن میٚے ییٚتہِ؟“ ورقہَن ووٚن: ”آ، یُس تہِ اِنسان توہہِ یِہ پَیغام ہؠتھ آو، تَمس سۭتۍ آو دُشمَنی کَرنہٕ۔ اَگر بہٕ تَتھ دۄہَس تام زِندٕ روزُس، بہٕ کَرٕ توہہِ پوٗرٕ طاقَتہٕ سۭتۍ مَدَد۔“ پَتہٕ گوٚو وَرَقہٕ تہِ یہِ دُنیا ترٛاوِتھ، تہٕ وَحی نازِل سَپدُن روٗد کینہہ وَختہٕ خٲطرٕ بَند۔

## حدیث 4 — وَحی دوبارٕ شروٗع گَژھُن

**راوی:** حضرت جابِر بِن عبداللہ رضی اللہ عنہ  
**حیثیت:** پہلے منظور شدہ

حَضرت جابِر بِن عبداللہ رضی اللہ عنہ چھِ روایت کَران زِ رسول اللہ صلی اللہ علیہ وسلمَن فَرموو: ”بہٕ اوسُس پَکان زِ اَچانَک مےٚ بوٗز آسمان پؠٹھ آواز۔ مےٚ وُچھ آسمانَس کُن تہٕ مےٚ وُچھ سُے فِرِشتہٕ، یُس مےٚ غارِ حرا ہَس منٛز وُچھ، کُرسی پؠٹھ بِہِتھ آسمانَس تہٕ زَمیٖنَس دَرمیان۔ تَمس وُچھِتھ گوٚوس بہٕ خَوف زَدٕ۔ پَتہٕ کٔر مےٚ گَرٕ کُن تَشریٖف تہٕ ووٚنُم: ’مےٚ وَلِیو! مےٚ وَلِیو!‘

پَتہٕ کٔر اللہ تعالیٰن یِم آیتہٕ نازِل: يَا أَيُّهَا الْمُدَّثِّرُ ۝ قُمْ فَأَنذِرْ ۝ وَرَبَّكَ فَكَبِّرْ ۝ وَثِيَابَكَ فَطَهِّرْ ۝ وَالرُّجْزَ فَاهْجُرْ۔ ’اے کَملی منٛز وَلِمَتؠو!—یعنی محمد صلی اللہ علیہ وسلم—وۄتھِو تہٕ خَبردار کَرِو، پَننِس رَب سٕنٛز بُزَرگی کَرِو بَیان، تہٕ پَنٕنۍ پَلَو تھاوِو صاف، تہٕ گَنٛدگی نِش روزِو دوٗر۔‘ اَمہِ پتہٕ لَج وَحی زور سان یِنۍ۔“

## حدیث 5 — قرآن جلدی یاد کَرنہٕ خٲطرٕ زبان ہَلاوُن

**راوی:** سعید بِن جُبیر، حضرت اِبنِ عباس رضی اللہ عنہما نِش  
**حیثیت:** صارف سٕنٛز جملہ بہ جملہ نظرِ ثانی مکمل — منظور شدہ

سعید بِن جُبیر چھِ روایت کَران زِ حضرت اِبنِ عباس رضی اللہ عنہمان اللہ تعالیٰ سُنٛدِس یَتھ فرمانَس مُتعلِق ووٚن: ”یِہ قرآن یاد تھاونہٕ موٗجوٗب مَہ ہَلاوِو تُہۍ پَنٕنۍ زَبان۔ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن ٲس وَحی نازِل گَژھنہٕ وِزِ زَبردست مُشکِلات سَپدان۔ تِم ٲس وَحی یاد تھاونہٕ موٗجوٗب پَنٕنۍ ہۄٹھ ہَلاوان۔“

حضرت اِبنِ عباس رضی اللہ عنہمان پَنٕنۍ ہۄٹھ ہَلاوِتھ ووٚن: ”بہٕ چھُس توہہِ برونٛٹھ کَنۍ پَنٕنۍ ہۄٹھ تَتھے پٲٹھۍ ہَلاوان، یِتھ کٔنۍ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن ہَلاوان ٲس۔“ سعیدَن رضی اللہ عنہَن تہِ ووٚن، پَنٕنۍ وۄٹھ ہَلاوِتھ: ”بہٕ چھُس پَنٕنۍ وۄٹھ تَتھے پٲٹھۍ ہَلاوان، یِتھ کٔنۍ مےٚ حضرت اِبنِ عباس رضی اللہ عنہما ہَلاوان وُچھ۔“

پَتہٕ کٔر اللہ تعالیٰن یِہ آیت نازِل: لَا تُحَرِّكْ بِهِ لِسَانَكَ لِتَعْجَلَ بِهِ ۝ إِنَّ عَلَيْنَا جَمْعَهُ وَقُرْآنَهُ۔ ”یِہ قرآن یاد تھاونہٕ موٗجوٗب مَہ ہَلاوِو تُہۍ پَنٕنۍ زَبان؛ تہٕ توہہِ یِہ یاد تھاوناوُن چھُ اَسہِ ذِمّہٕ۔“

اللہ تعالیٰن فَرموو: فَإِذَا قَرَأْنَاهُ فَاتَّبِعْ قُرْآنَهُ۔ ”پَس ییلہِ اَسۍ یِہ پَرَو، تیلہِ کٔرِو تَمیٚچ تِلاوَت وارٕ وارٕ۔“ حضرت اِبنِ عباس رضی اللہ عنہمان ووٚن زِ یَمیُک مطلب چھُ: ”غورٕ سان بوٗزِو تہٕ ژھۄپ روزِو۔“ اللہ تعالیٰن پَتہٕ فَرموو: ثُمَّ إِنَّ عَلَيْنَا بَيَانَهُ۔ ”پَتہٕ تَمیُک بیان کَرُن تہِ چھُ اَسہِ ذِمّہٕ۔“

اَمہِ پتہٕ ییلہِ فِرِشتہٕ جبرئیل علیہ السلام رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن نِش یِوان ٲس، تِم ٲس غورٕ سان بوٗزان؛ تہٕ ییلہِ جبرئیل علیہ السلام واپس گَژھان ٲس، تیلہِ ٲس رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّم تَتھے پٲٹھۍ پَران، یِتھ کٔنۍ جبرئیل علیہ السلامَن پَرِتھ اوس۔

**راوی:** حضرت اِبنِ عباس رضی اللہ عنہما  
**حیثیت:** صارف سٕنٛز جملہ بہ جملہ نظرِ ثانی مکمل — منظور شدہ

حضرت اِبنِ عباس رضی اللہ عنہما چھِ روایت کَران: ”رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّم ٲس سٲریو اِنسانَن کھۄتہٕ زیادٕ سَخی۔ رمضانَس منٛز ٲس تِمَن سٕنز سخاوت ساروِے کھۄتہٕ زیادٕ گَژھان، ییلہِ فِرِشتہٕ جبرئیل علیہ السلام تِمَن نِش یِوان ٲس۔ فِرِشتہٕ جبرئیل علیہ السلام ٲس رمضانَس پرٛتھ رٲژِ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن نِش یِوان تہٕ تِمَن سۭتۍ قرآنِ کریم دَہراوان۔ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّم ٲس سوزنہٕ آمٕژِ ہواہِ کھۄتہٕ تہِ زیادٕ سَخی، خیرات کَرنَس منٛز۔“

## حدیث 7 — ابو سُفیان تہٕ ہِرقل

**راوی:** حضرت اِبنِ عباس رضی اللہ عنہما، ابو سُفیان بِن حرب نِش  
**حیثیت:** صارف سٕنٛز جملہ بہ جملہ نظرِ ثانی مکمل — منظور شدہ

حضرت اِبنِ عباس رضی اللہ عنہما چھِ روایت کَران زِ ابو سُفیان بِن حربَن ووٚن: ہِرقلَن سوز مےٚ نِش اَکھ قاصِد، تَمہِ وِزِ ییلہِ بہٕ قُریشَن ہِنٛدِس اَکِس تجارتی قافِلَس سۭتۍ شامَس منٛز اوسُس۔ یِہ کَتھ چھِ تَمہِ وَقتُک ییلہِ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن تہٕ ابو سُفیانَس تہٕ قُریشَن ہٕنٛدؠن کافِرَن دَرمیان صُلح ٲس۔ ابو سُفیان تہٕ تَمہِ سُنٛدۍ ساتھی گٔیہِ ایلیا—یروٗشلم—منٛز ہِرقلَس نِش۔ ہِرقلَن بُلاو تِم دَربارَس منٛز، یَتہِ زوٚپٲر رومَن ہٕنٛدۍ بٔڈۍ بٔڈۍ سردار ٲسۍ۔ تَمہِ بُلاو پَنُن تَرجُمان تہٕ پَرژھُن: ”توہہِ منٛز کُس چھُ تَتھ مَردَس سٲریو کھۄتہٕ نزدیٖک رِشتہٕ دار، یُس پَنُن پان نَبی وَنان چھُ؟“ ابو سُفیانَن ووٚن: ”اَتھ جَماعَتَس منٛز چھُس بہٕ تَمس سٲریو کھۄتہٕ نزدیٖک رِشتہٕ دار۔“

ہِرقلَن دیُت حُکم: ”یِہ ابو سُفیان کٔرہوٗن مےٚ نزدیٖک تہٕ اَمہِ سٕندۍ ساتھی تھاوِہو اَمیِس پَتہٕ۔“ پَتہٕ ووٚنُن تَرجُمانَس: ”بہٕ چھُس رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَس مُتعلِق اَمیِس سَوال کَران؛ اَگر یِہ اَپوٗز وَنِ، تیلہِ کٔرِو اَمیِچ تَردیٖد۔“ ابو سُفیانَن ووٚن: ”خُدایہِ سُنٛد قَسَم، اَگر مےٚ پَننؠن یارَن نِش اَپوٗز وَنن وول بَدنٲمی ہُنٛد خوف نہٕ آسہٕ ہا، بہٕ وَنہٕ ہا تَمس مُتعلِق اَپوٗز۔“

ہِرقلَن گوٚڈٕ پرژھُم: ”تُہۍ منٛز کیا چھِ تَمہِ سٕندِ خاندانٕچ حیثیت؟“ مےٚ ووٚن: ”سُہ چھُ عزت دار خاندانُک۔“ تَمہِ پرژھ: ”یَمہِ برونٛہہ چھا توہہِ منٛز کانٛسہِ بَنومُت پَنُن پان نَبی؟“ مےٚ ووٚن: ”نہٕ۔“ تَمہِ پرژھ: ”تَمہِ سٕندؠن بُزرگَن منٛز اوسوا کانٛہہ بادشاہ؟“ مےٚ ووٚن: ”نہٕ۔“ تَمہِ پرژھ: ”تَمہِ پَتہٕ چھا شریٖف تہٕ اَمیٖر لُکھ گَژھان، کِنہٕ غریٖب؟“ مےٚ ووٚن: ”غریٖب لُکھ۔“ تَمہِ پرژھ: ”تَمہِ سُنٛدۍ پَیروکار چھا بڑان کِنہٕ گھَٹان؟“ مےٚ ووٚن: ”بڑان۔“ تَمہِ پرژھ: ”کیا تَمہِ سُنٛد دیٖن قبوٗل کٔرِتھ چھُ کانٛہہ اَمیِہ نِش ناخوش گَژھان تہٕ دیٖن ترٛاوان؟“ مےٚ ووٚن: ”نہٕ۔“ تَمہِ پرژھ: ”نَبوّتُک دعویٰ کَرنہٕ برونٛہہ کٔروا توہہِ زانہہ تَمس اَپوٗز وَننُک اِلزام؟“ مےٚ ووٚن: ”نہٕ۔“ تَمہِ پرژھ: ”کیا سُہ چھُ عہد شِکنی کَران؟“ مےٚ ووٚن: ”نہٕ۔ وۄنۍ چھُ اَسہِ تَمس سۭتۍ صُلح، تہٕ اَسہِ چھُنہٕ پَے زِ سُہ اَتھ دَورانَس منٛز کیا کَرِ۔“ ابو سُفیانَن ووٚن زِ یَمہِ یَکۍ جُملہٕ علاوٕ میول نہٕ مےٚ تَمس خِلاف کینٛہہ وَننُک موقَعہٕ۔

ہِرقلَن پرژھ: ”کیا توہہِ تہٕ تَمس دَرمیان سَپدوا زانہہ جَنگ؟“ مےٚ ووٚن: ”آ۔“ تَمہِ پرژھ: ”تُہُنٛد جَنگُک نتیٖجہٕ کِتھ پٲٹھۍ روٗد؟“ مےٚ ووٚن: ”کُنہِ ساتہٕ چھُ سُہ اَسہِ پؠٹھ غٲلِب گَژھان تہٕ کُنہِ ساتہٕ اَسۍ تَمس پؠٹھ۔“ تَمہِ پرژھ: ”سُہ کیا چھُ توہہِ حُکم دِوان؟“ مےٚ ووٚن: ”سُہ چھُ اَسہِ وَنان زِ صِرف اللہ تعالیٰ سٕنز عبادت کٔرِو، تَمس سۭتۍ کانٛہہ شریٖک مٲ تھاوِو، تہٕ یِم کَتھہٕ تُہٕنٛدۍ بُزرگ وَنان ٲسۍ تِم ترٛاوِو۔ سُہ چھُ اَسہِ نَماز قٲیم کَرنُک، پوٗز وَننُک، پاک دامن روزنُک تہٕ رِشتہٕ دارَن سۭتۍ اَصٕل سُلوٗک کَرنُک حُکم دِوان۔“

پَتہٕ ووٚن ہِرقلَن تَرجُمانَس: ”یِمس وَن: مےٚ پرژھٕے ژےٚ تَمہِ سُنٛد خاندان تہٕ ژےٚ ووٚنُتھ زِ سُہ چھُ شریٖف خاندانَس منٛزٕ۔ تَمام رَسوٗل چھِ پَننؠن قَومَن ہٕنٛدؠن شریٖف خاندانَن منٛز پَیدا گژھان۔ مےٚ پرژھٕے ژےٚ زِ اَمہِ برونٛہہ کوٚروا کانٛہہ یِژھ دَعوت؛ ژےٚ ووٚنُتھ نہٕ۔ اَگر کانٛہہ کٔرِتھ آسہٕ ہا، بہٕ سَمجہٕ ہا زِ یِہ چھُ اَکھ گوٚڈلی کَتھ دَہراوان۔ مےٚ پرژھٕے ژےٚ زِ تَمہِ سُنٛدؠن بُزرگَن منٛز اوسُکھا کانٛہہ بادشاہ؛ ژےٚ ووٚنُتھ نہٕ۔ اَگر آسہٕ ہا، بہٕ سَمجہٕ ہا زِ یِہ چھُ پَننؠن بُزرگَن ہُنٛد مُلک واپس حٲصِل کَرُن یَژھان۔

”مےٚ پرژھٕے ژےٚ زِ دعویٰ کَرنہٕ برونٛہہ کٔروا توہہِ تَمس اَپوٗز وَننُک اِلزام؛ ژےٚ ووٚنُتھ نہٕ۔ پَس مےٚ سَمجھ آو زِ یُس اِنسان لُکَن مُتعلِق اَپوٗز چھُنہٕ وَنان، سُہ کِتھ کٔنۍ ہؠکہِ اللہ تعالیٰ سُنٛدِس ناوَس پؠٹھ اَپوٗز وَنِتھ؟ مےٚ پرژھٕے ژےٚ زِ تَمس پَتہٕ چھا اَمیٖر لُکھ گَژھان کِنہٕ غریٖب؛ ژےٚ ووٚنُتھ غریٖب۔ رَسوٗلَن ہٕنٛدۍ پَیروکار چھِ ہمیشہٕ یِمے لُکھ ٲسمٕتۍ۔ مےٚ پرژھٕے ژےٚ زِ تَمہِ سُنٛدۍ پَیروکار چھا بڑان کِنہٕ گھَٹان؛ ژےٚ ووٚنُتھ بڑان۔ پوٗز دیٖن چھُ تَتھے پٲٹھۍ بڑان، یوتام سُہ مکمّل گَژھِ۔

”مےٚ پرژھٕے ژےٚ زِ تَمہِ سُنٛد دیٖن قبوٗل کٔرِتھ چھُ کانٛہہ ناخوش گَژھان تہٕ دیٖن ترٛاوان؛ ژےٚ ووٚنُتھ نہٕ۔ پوٗز ایمان ییلہِ دِلَس منٛز داخِل گژھان تہٕ تَمس سۭتۍ رَلان چھُ، تیلہِ چھُ تَمیُک حال یِتھے آسان۔ مےٚ پرژھٕے ژےٚ زِ سُہ چھا عہد شِکنی کَران؛ ژےٚ ووٚنُتھ نہٕ۔ رَسوٗل چھِ نہٕ عہد شِکنی کَران۔ پَتہٕ مےٚ پرژھٕے ژےٚ زِ سُہ چھُ توہہِ کیا حُکم دِوان۔ ژےٚ ووٚنُتھ زِ سُہ چھُ توہہِ صِرف اللہ تعالیٰ سٕنز عبادت کَرنُک، تَمس سۭتۍ کانٛہہ شریٖک نہٕ تھاونُک، بُت پَرستی ترٛاونُک، نَماز قٲیم کَرنُک، پوٗز وَننُک تہٕ پاک دامن روزنُک حُکم دِوان۔

”اَگر ژےٚ پوٗز ووٚنُتھ، یِہ نَبی کَرِ جلدے یَتھ جایہِ پؠٹھ تہِ حُکوٗمت، یَتھ میٲنۍ حُکوٗمت چھِ۔ مےٚ اوس اِلٰہی کِتابَن منٛز پَے زِ اَکھ نَبی چھُ یِنہٕ وول، مگر مےٚ اوس نہٕ خیال زِ سُہ آسِ توہہِ منٛزٕ۔ اَگر مےٚ یَقیٖن آسہٕ ہا زِ بہٕ واتہٕ تَمس نِش، بہٕ گَژھہٕ ہا فوٚری تَمس وُچھنہٕ؛ تہٕ اَگر بہٕ تَمس نِش آسہٕ ہا، بہٕ چھلہٕ ہا تَمہِ سٕندۍ کھور۔“

پَتہٕ اَناو ہِرقلَن سُہ خَط، یُس رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن دِحیہ کلبی رضی اللہ عنہَس دِتھ بُصریٰ ہُندِس حاکِمَس نِش سوزمُت اوس تہٕ تَمہِ ہِرقلَس پَرنہٕ خٲطرٕ سوز۔ خَطَس منٛز اوس لیٚکھِتھ:

”بسم اللہ الرحمٰن الرحیم۔ اللہ تعالیٰ سُنٛدِس بَنٛدَس تہٕ رَسوٗلَس محمدَس طرفہٕ، رومَن ہُنٛدِس بادشاہ ہِرقلَس ناو۔ سلام تَتھ اِنسانَس پؠٹھ، یُس ہِدایت رَٹِ۔ اَمہِ پتہٕ: بہٕ چھُس ژےٚ اِسلامَس کُن دَعوت دِوان۔ اِسلام قبوٗل کَر، سلامت روزِکھ تہٕ اللہ تعالیٰ دِیِکھ ژےٚ دۄگنہٕ اَجر۔ اَگر ژٕ یِہ دَعوت رَد کَرِکھ، تیلہِ آسِ ژےٚ پَننؠن رِعایَتَن ہُنٛد گُناہ تہِ۔ اے اَہلِ کِتاب! یِو تَتھ اَکِس کَلمَس کُن یُس اَسہِ تہٕ توہہِ دَرمیان یَکسان چھُ: اَسۍ کَرَو نہٕ عبادت مگر اللہ تعالیٰ سٕنز، اَسۍ تھاوَو نہٕ تَمس سۭتۍ کانٛہہ شریٖک، تہٕ اَسہِ منٛز بَنایِہ نہٕ کانٛہہ اَکِس تہِ اللہ تعالیٰ سوا شریٖک رَبَس۔ اَگر تِم پِھرَن، تیلہِ وَنِو: گواہ روزِو زِ اَسۍ چھِ مُسلمان۔“

ابو سُفیانَن ووٚن: ”ییلہِ ہِرقلَن پَنٕنۍ کَتھ مکلاو تہٕ خَط پَر، دَربارَس منٛز گوٚو بوٚڈ شور۔ اَسۍ آیہِ دَربارَس منٛزٕ نیبَر کَڑنہٕ۔ مےٚ ووٚن پَننؠن یارَن زِ ابو کبشہ سُنٛدِس پُترَس سٕنٛز کَتھ گٔیہِ اِتھ کَنۍ طاقت وَلی، زِ رومَن ہُنٛد بادشاہ تہِ چھُ تَمس نِش کھوژان۔ تَمہِ دۄہَس پؠٹھ ووت مےٚ یَقیٖن بڑُن زِ سُہ گَژھِ جلدے غٲلِب، یوتام بہٕ اللہ تعالیٰن اِسلامَس منٛز داخِل کوٚرُس۔“

اِبنِ ناطورَن بیان کوٚر زِ ہِرقل ییلہِ ایلیا منٛز اوس، اَکِس صُبحَس وۄتھ سُہ پریشان۔ تَمہِ سٕندؠن کینہہ پادریَو ووٚنُس: ”اَز کٲژِ چھُکھ پریشان؟“ ہِرقل اوس نَجوٗمی تہٕ سِتارَن ہُنٛد عِلم زانان۔ تَمہِ ووٚن: ”مےٚ وُچھ راتَس سِتارَن کُن؛ خَتنہٕ کَرن والؠن ہُنٛد بادشاہ چھُ ظٲہِر گومُت۔ دُنیاہَس منٛز چھِ خَتنہٕ کَرناوان کَم؟“ تِمو ووٚن: ”یَہوٗدیَن علاوٕ چھُنہٕ کانٛہہ خَتنہٕ کَران؛ پَننِس مُلکَس منٛز یوتاہ تہِ یَہوٗدی چھِ، تِمَن مَرنُک حُکم دِہ۔“

تِم یِہ کَتھ کَران ٲسۍ زِ غَسّان ہُنٛدِس بادشاہَس طرفہٕ آو اَکھ قاصِد، یُس رَسوٗلُ اللہ صلی اللہ علیہ وسلم سٕنز خبر ہؠتھ آمُت اوس۔ ہِرقلَن حُکم دِت زِ وُچھِو، یَمیِس چھا خَتنہٕ گژھِتھ کِنہٕ نہٕ۔ تِمو یَمیِس وُچھِتھ ہِرقلَس ووٚن زِ آ، یَمیِس چھِ خَتنہٕ گژھِتھ۔ ہِرقلَن پرژھ عربَن مُتعلِق؛ قاصِدَن ووٚن زِ عرب تہِ چھِ خَتنہٕ کَران۔ تیلہِ ووٚن ہِرقلَن: ”یِمَن لُکَن ہُنٛد بادشاہ چھُ ظٲہِر گومُت۔“

پَتہٕ لیوٚکھ ہِرقلَن رومَس منٛز پَننِس اَکِس عِلم وٲلِس دوستَس خَط۔ سُہ گَو حِمصَس تہٕ روٗد تَتہِ پَننِس دوستَس سُنٛد جواب یِنہٕ تام۔ دوستَن کٔر تَمہِ سٕنز راے تٲیید زِ نَبی صلی اللہ علیہ وسلم چھِ ظٲہِر گٲمٕتۍ تہٕ تِم چھِ پوٗز نَبی۔ پَتہٕ بُلاو ہِرقلَن رومَن ہٕنٛدۍ سٲری سردار حِمصَس پَننِس مَحَلَس منٛز تہٕ دَروازٕ کٔر بَنٛد۔ پَتہٕ ووٚن: ”اے رومؠو! اَگر توہہ کامیٲبی، ہِدایت تہٕ پَنٕنۍ بادشاہت قٲیم روزُن یَژھان چھِو، تیلہِ کَرِو یَتھ نَبیَس بیعت تہٕ قبوٗل کَرِو اِسلام۔“

یِہ بوٗزِتھ وٲتھ تِم دَروازَن کُن، مگر دَروازٕ ٲسۍ بَنٛد۔ ہِرقلَن ییلہِ تِمن ہُنٛد اِسلامَس خِلاف نَفرت وُچھ تہٕ تِمن ہُنٛد اِسلام قبوٗل کَرنُک اُمید مۄکل، حُکم دِت زِ تِم واپس اَنِو۔ ییلہِ تِم واپس آیہِ، تَمہِ ووٚن: ”مےٚ یُس کینٛہہ ووٚن، سُہ اوس صِرف تُہٕنٛد دیٖن پؠٹھ مَضبوٗطی آزماونہٕ خٲطرٕ؛ وۄنۍ وُچھ مےٚ سُہ۔“ پَتہٕ کٔر تِمو تَمس سَجدٕ تہٕ تِم گٔیہِ تَمس پؠٹھ خوش۔ یَتھ سۭتۍ گوٚو ہِرقل سُنٛد قصّہٕ خَتم۔

## حدیث 18 — عَقبہٕچ بیعت

**راوی:** حضرت عُبادہ بِن صامت رضی اللہ عنہ  
**حیثیت:** صارف سٕنٛز نظرِ ثانی مکمل — منظور شدہ

حضرت عُبادہ بِن صامت رضی اللہ عنہ، یِمو بَدر کِس جَنگَس مَنٛز حصہٕ نیوٗ تہٕ اوس نَقیٖب—شیٚن ہٕندِس جَمٲژ ہُنٛد رَہنما—عَقبہٕ عہد کِس رٲژ مَنٛز، چھِ روایت کَران:

رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَن فَرموو، تَمہِ وِزِ ییلہِ صحابہَن ہُنٛد اَکھ جماعت تِمَن زوٚپٲر ٲس: ”تُہۍ کَرِو مےٚ سۭتۍ یِمَن کَتھَن پؠٹھ بیعت: اللہ تعالہس سۭتۍ کانٛہہ شریٖک مَہ تھاوِو؛ ژور مَہ کَرِو؛ زِنا مَہ کَرِو؛ پَنٕنۍ شُرۍ مَہ مارِو؛ کٲنسہِ شَخصَس پؠٹھ مَہ لَگٲوِو اِلزام، یُس بے گُناہ آسہِ، یا اَمہِ قٕسمٕچ اَفواہ پھٲلٲوِو؛ تہٕ جان کٲم (good deed) کَرنَس منٛز نافرمانی مَہ کَرِو۔“

پَتہٕ فَرموو: ”توہہِ منٛز یُس یِہ عہد پوٗرٕ کَرِ، تَمس دِیِہ اللہ تعالہ اَجر۔ یُس یِموو منٛز کانٛہہ گُناہ کَرِ تہٕ میلہِس دُنیاہَس منٛز اَمیُک سزا، سۄ سزا بَنِ تَمہِ گُناہُک کفارہٕ۔ تہٕ یُس یِموو منٛز کانٛہہ گُناہ کَرِ تہٕ اللہ تعالہ تھاوِ تَمسُنٛد گُناہ ژوٗرِ، تیلہِ چھُ تَمیُک فیصلہٕ اللہ تعالہ سٕندِ اختیارَس منٛز؛ چاہے تَمس معاف کَرِ یا سزا دِیِس۔“

حضرت عُبادہ بِن صامت رضی اللہ عنہَن ووٚن: ”پَتہٕ کٔر اَسہِ یِمَن کَتھَن پؠٹھ رَسوٗلُ اللہ صَلَّی اللہُ عَلَیہِ وَسَلَّمَس بیعت۔“
